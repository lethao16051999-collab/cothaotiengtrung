function nhomApi_(e) {
  var b;
  try { b = JSON.parse(e.postData.contents); } catch (x) { return null; }
  if (!b || ["nlop", "ngui", "nkq", "ncho", "nnx", "ndat", "nghep", "nroi"].indexOf(b.a) < 0) return null;
  var out;
  try { out = nhRun_(b); } catch (x) { out = { ok: false, err: "loi", msg: String(x) }; }
  out.nh = 1;
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function nhRun_(b) {
  if (b.a == "nlop") return nhLop_(b);
  if (b.a == "nkq") return nhKq_(b);
  if (b.a == "ngui") return nhLock_(function () { return nhGui_(b); });
  if (b.a == "nghep") return gcGhep_(b);
  if (b.a == "ndat") return nhLock_(function () { return gcDat_(b); });
  if (b.a == "nroi") return nhLock_(function () { return gcRoi_(b); });
  if (!nhPass_(b.pass)) return { ok: false, err: "pass" };
  if (b.a == "ncho") return nhCho_(b);
  if (b.a == "nnx") return nhLock_(function () { return nhNx_(b); });
}

function nhLock_(f) { var l = LockService.getScriptLock(); l.waitLock(20000); try { return f(); } finally { l.releaseLock(); } }
function nhPass_(p) {
  var sh = SpreadsheetApp.getActive().getSheetByName("Cài đặt");
  var v = sh ? String(sh.getRange("B1").getValue()).trim() : "";
  return !!v && String(p || "").trim() === v;
}
function nhTxt_(s) { s = String(s == null ? "" : s); return /^[=+\-@]/.test(s) ? "'" + s : s; }
function nhNow_() { return Utilities.formatDate(new Date(), "Asia/Ho_Chi_Minh", "dd/MM/yyyy HH:mm"); }

/* danh sách học viên: {MÃ: {code, name, cls, st}} */
function nhHvAll_() {
  var sh = SpreadsheetApp.getActive().getSheetByName("Học viên"), m = {};
  if (!sh || sh.getLastRow() < 2) return m;
  var d = sh.getDataRange().getValues(), h = d[0].map(function (x) { return String(x).trim(); });
  var iM = h.indexOf("Mã"), iT = h.indexOf("Họ tên"), iL = h.indexOf("Lớp"), iS = h.indexOf("Trạng thái");
  if (iM < 0) iM = 0; if (iT < 0) iT = 1; if (iL < 0) iL = 2;
  for (var i = 1; i < d.length; i++) {
    var c = String(d[i][iM]).trim().toUpperCase();
    if (c) m[c] = { code: c, name: String(d[i][iT]).replace(/\s+/g, " ").trim(), cls: String(d[i][iL]).replace(/\s+/g, " ").trim(), st: iS >= 0 ? String(d[i][iS]).trim() : "" };
  }
  return m;
}

/* học viên đã nhập mã: lấy danh sách bạn cùng lớp để chọn vào nhóm */
function nhLop_(b) {
  var all = nhHvAll_(), me = all[String(b.code || "").trim().toUpperCase()];
  if (!me) return { ok: false, err: "no_code" };
  var hv = [];
  Object.keys(all).forEach(function (k) { var h = all[k]; if (h.cls == me.cls && h.st != "Nghỉ") hv.push({ code: h.code, name: h.name }); });
  hv.sort(function (x, y) { return x.name.split(" ").pop().localeCompare(y.name.split(" ").pop(), "vi"); });
  /* bạn đã ở trong cặp / nhóm hôm nay: {mã: {pid, by}} */
  var bz = {};
  gcToday_(gcRows_(1)).forEach(function (it) { it.js.m.forEach(function (x) { if (x.c) bz[x.c] = { pid: it.pid, by: it.js.by, byc: it.js.byc }; }); });
  return { ok: true, v2: 1, cls: me.cls, hv: hv, bz: bz };
}

/* nhóm gửi (hoặc cập nhật) thông tin: tạo / ghi đè các dòng cùng Mã nhóm, giữ nguyên nhận xét của cô */
function nhGui_(b) {
  var g = b.g;
  if (!g || !g.gid || !g.m || !g.m.length) return { ok: false, err: "data" };
  var all = nhHvAll_(), cls = String(g.cls || "");
  g.m.forEach(function (x) { var h = x.c && all[String(x.c).toUpperCase()]; if (h) { x.n = h.name; if (!cls) cls = h.cls; } });
  var R = nhRows_(), i0 = -1, n = 0;
  R.d.forEach(function (r, i) { if (r[0] == g.gid) { if (i0 < 0) i0 = i; n++; } });
  var old = i0 >= 0 ? R.d[i0] : null, ev = old ? old.slice(13, 22) : ["", "", "", "", "", "", "", "", ""];
  var at = old ? old[1] : nhNow_(), ms = old ? old[22] : Date.now();
  var ca = g.m.map(function (x) { return x.n + (x.r ? " (" + x.r + ")" : ""); }).join(", ");
  var ow = g.ow ? 1 : 0, dev = String(g.dev || "").toUpperCase();
  var js = JSON.stringify({ m: g.m, set: g.set, zh: g.zh, dev: g.dev || "", ow: ow, pid: g.pid || "" });
  /* mỗi bạn một máy (ow): báo cáo và trạng thái ghi riêng vào dòng của bạn gửi, dòng các bạn khác giữ nguyên */
  var cu = {};
  if (ow) R.d.forEach(function (r) { if (r[0] == g.gid) cu[r[8] ? String(r[8]) : "n:" + r[9]] = { rep: r[12], stt: r[11] }; });
  var rows = g.m.map(function (x) {
    var key = x.c ? String(x.c).toUpperCase() : (x.t ? "GV" : "n:" + x.n), mine = !ow || (x.c && String(x.c).toUpperCase() == dev), o = cu[key] || {};
    return [g.gid, at, nhTxt_(cls), g.lv, nhTxt_(g.st), g.tid, nhTxt_(g.tt), nhTxt_(ca), x.c || (x.t ? "GV" : ""), nhTxt_(x.n), x.r || "",
      mine ? (g.stt || "Đang làm") : (o.stt || "Đang làm"), nhTxt_(mine ? (g.rep || "") : (o.rep || ""))].concat(ev).concat([ms, js]);
  });
  if (g.pid) gcTouch_(g);
  if (i0 >= 0 && n == rows.length) R.sh.getRange(i0 + 2, 1, n, NH_HD.length).setValues(rows);
  else {
    if (i0 >= 0) R.sh.deleteRows(i0 + 2, n);
    R.sh.getRange(R.sh.getLastRow() + 1, 1, rows.length, NH_HD.length).setValues(rows);
  }
  return { ok: true, cls: cls };
}

  if (!bm.some(function (v) { return v != null; }) && !r[20]) return null;
  return { bm: bm, nx: String(r[20] || ""), at: String(r[21] || "") };
}

  nhRows_().d.forEach(function (r) { if (want[r[0]] && !kq[r[0]]) { var e = nhEv_(r); if (e) kq[r[0]] = e; } });

}

/* cô xem danh sách nhóm */
function nhCho_(b) {
  var from = Date.now() - (b.days || 30) * 864e5, G = {}, list = [];
  nhRows_().d.forEach(function (r) {
    if (+r[22] < from) return;
    var g = G[r[0]];
    if (!g) {
      var js = {}; try { js = JSON.parse(r[23] || "{}"); } catch (x) {}
      g = G[r[0]] = { gid: r[0], at: +r[22], t: r[1], cls: r[2], lv: +r[3], st: r[4], tid: r[5], tt: r[6], zh: js.zh || "", stt: r[11], rep: r[12], m: [], ev: nhEv_(r), ow: js.ow ? 1 : 0, _r: [], _n: 0, _d: 0 };
      list.push(g);
    }
    g.m.push({ c: r[8] == "GV" ? "" : r[8], n: r[9], r: r[10], t: r[8] == "GV" ? 1 : 0 });
    if (g.ow) {
      if (String(r[12]).trim()) g._r.push(r[9] + ": " + r[12]);
      if (r[8] && r[8] != "GV") { g._n++; if (r[11] == "Xong") g._d++; }
    }
  });
  list.forEach(function (g) {
    if (g.ow) { g.rep = g._r.join("\n\n"); g.stt = g._n && g._d == g._n ? "Xong" : g._d ? "Đang làm · " + g._d + "/" + g._n + " bạn xong" : "Đang làm"; }
    delete g._r; delete g._n; delete g._d;
  });
  if (list.length > 800) list = list.slice(-800);
  return { ok: true, v2: 1, g: list };
}

/* cô lưu nhận xét: bm = 6 số (0 Chưa đạt, 1 Đạt, 2 Tốt, null chưa chấm), nx = chú thích */
function nhNx_(b) {
  var R = nhRows_(), n = 0, now = nhNow_(), bm = (b.bm || []).slice(0, 6), tot = 0, has = 0;
  while (bm.length < 6) bm.push(null);
  var txt = bm.map(function (v) { if (v == null || v === "") return ""; has++; tot += +v; return NH_LV[+v]; });
  var val = [txt.concat([has ? tot + "/" + (has * 2) : "", nhTxt_(b.nx || ""), now])];
  R.d.forEach(function (r, i) { if (r[0] == b.gid) { R.sh.getRange(i + 2, 14, 1, 9).setValues(val); n++; } });
  return { ok: n > 0, n: n, at: now };
}

/* ===== Ghép cặp mỗi bạn một máy (04/11/2026) =====

   Máy các bạn trong cặp hỏi "nghep" → nhận cặp, vai, thẻ đang làm. Bạn đã ở trong cặp hôm nay thì không ai chọn được nữa. */
var GC_SH = "Ghép cặp";
var GC_HD = ["Ngày", "Mã cặp", "Lớp", "Cấp", "Người lập", "Thành viên", "Mã các bạn", "Trạng thái", "Bộ ôn", "Thẻ", "Nhiệm vụ", "Mã nhóm", "Người bốc", "Cập nhật", "ms", "JSON"];
function gcDay_() { return Utilities.formatDate(new Date(), "Asia/Ho_Chi_Minh", "yyyy-MM-dd"); }
function gcD_(v) { return v instanceof Date ? Utilities.formatDate(v, "Asia/Ho_Chi_Minh", "yyyy-MM-dd") : String(v); }
function gcSheet_() {

function gcRows_(ro) {

  if (!sh) return { sh: null, d: [] };

/* các cặp còn hiệu lực hôm nay */
function gcToday_(R) {
  var day = gcDay_(), L = [];
  R.d.forEach(function (r, i) {
    if (gcD_(r[0]) != day || r[7] != "Đang ghép") return;
    var js = {}; try { js = JSON.parse(r[15] || "{}"); } catch (x) {}
    if (!js.m) js.m = [];
    L.push({ i: i, r: r, pid: String(r[1]), js: js });
  });
  return L;
}
function gcHas_(it, c) { return it.js.m.some(function (x) { return x.c && x.c == c; }); }
function gcWrite_(R, it) {
  var js = it.js, m = js.m;
  var row = [gcDay_(), it.pid, nhTxt_(js.cls || ""), js.lv || "", nhTxt_(js.by || ""), nhTxt_(m.map(function (x) { return x.n + (x.r ? " (" + x.r + ")" : ""); }).join(", ")),
    "|" + m.filter(function (x) { return x.c; }).map(function (x) { return x.c; }).join("|") + "|", it.stt || "Đang ghép",
    nhTxt_(js.set || ""), js.tid || "", nhTxt_(js.tt || ""), js.gid || "", js.who || "", nhNow_(), String(Date.now()), JSON.stringify(js)];
  if (it.i >= 0) R.sh.getRange(it.i + 2, 1, 1, GC_HD.length).setValues([row]);
  else R.sh.getRange(R.sh.getLastRow() + 1, 1, 1, GC_HD.length).setValues([row]);
}
/* lập / sửa cặp: b = {code, lv, pid?, m:[{n,c,r,t}]} */
function gcDat_(b) {
  var all = nhHvAll_(), meC = String(b.code || "").trim().toUpperCase(), me = all[meC];
  if (!me) return { ok: false, err: "no_code" };
  var m = (b.m || []).slice(0, 12).map(function (x) {
    var c = String(x.c || "").trim().toUpperCase(), h = c && all[c];
    return { n: h ? h.name : String(x.n || "").replace(/\s+/g, " ").trim().slice(0, 60), c: h ? c : "", r: x.r == "B" ? "B" : x.r == "A" ? "A" : "", t: x.t ? 1 : 0 };
  }).filter(function (x) { return x.n; });
  if (!m.some(function (x) { return x.c == meC; })) m.unshift({ n: me.name, c: meC, r: "A", t: 0 });
  var R = gcRows_();
  if (R.d.length > 3000) { R.sh.deleteRows(2, 1500); R = gcRows_(); }
  var L = gcToday_(R), pid = String(b.pid || ""), bad = [];
  var cur = pid ? L.filter(function (it) { return it.pid == pid && gcHas_(it, meC); })[0] : null;
  if (!cur) pid = "";
  var mine = L.filter(function (it) { return it.pid != pid && gcHas_(it, meC); })[0];
  if (mine) return { ok: false, err: "me_busy", pid: mine.pid, by: mine.js.by || "" };
  m.forEach(function (x) {
    if (!x.c || x.c == meC) return;
    var o = L.filter(function (it) { return it.pid != pid && gcHas_(it, x.c); })[0];
    if (o) bad.push({ c: x.c, n: x.n, by: o.js.by || "" });
  });
  if (bad.length) return { ok: false, err: "busy", bad: bad };
  var it = cur || { i: -1, pid: "P" + Date.now().toString(36) + Math.floor(Math.random() * 1296).toString(36), js: { by: me.name, byc: meC } };
  it.js.m = m; it.js.cls = me.cls; it.js.lv = b.lv || it.js.lv || "";
  gcWrite_(R, it);
  return { ok: true, pid: it.pid, m: m, by: it.js.by, byc: it.js.byc };
}
/* rời cặp: người lập rời hoặc còn dưới 2 người thì hủy cả cặp */
function gcRoi_(b) {
  var c = String(b.code || "").trim().toUpperCase(), R = gcRows_(), L = gcToday_(R), n = 0;
  L.forEach(function (it) {
    if ((b.pid && it.pid != b.pid) || !gcHas_(it, c)) return;
    it.js.m = it.js.m.filter(function (x) { return x.c != c; });
    if (it.js.byc == c || it.js.m.length < 2) it.stt = "Đã hủy";
    gcWrite_(R, it); n++;
  });
  return { ok: true, n: n };
}
/* máy học viên hỏi: hôm nay em có trong cặp nào không, cặp đang làm thẻ nào */
function gcGhep_(b) {
  var c = String(b.code || "").trim().toUpperCase(), L = gcToday_(gcRows_(1)).filter(function (it) { return gcHas_(it, c); });
  L.sort(function (x, y) { return (+y.r[14]) - (+x.r[14]); });
  var it = L[0];
  if (!it) return { ok: true, v2: 1, p: null };
  var j = it.js;
  return { ok: true, v2: 1, p: { pid: it.pid, lv: +j.lv || 0, cls: j.cls || "", m: j.m, by: j.by || "", byc: j.byc || "", set: j.set || "", tid: j.tid || "", tt: j.tt || "", gid: j.gid || "", who: j.who || "" } };
}
/* nhóm vừa bốc / mở thẻ (ngui có pid): ghi thẻ đang làm vào cặp */
function gcTouch_(g) {
  var R = gcRows_(1); if (!R.sh) return;
  var it = gcToday_(R).filter(function (x) { return x.pid == g.pid; })[0];
  if (!it || (it.js.gid == g.gid && it.js.tid == g.tid)) return;
  it.js.set = g.set; it.js.tid = g.tid; it.js.tt = g.tt; it.js.gid = g.gid; it.js.who = String(g.dev || "").toUpperCase();
  if (g.lv) it.js.lv = g.lv;
  gcWrite_(R, it);
}
